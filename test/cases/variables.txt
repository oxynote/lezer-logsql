$env

==>

Query(Phrase(VariableReference))

# a variable as a field value

env:$env status:>$min

==>

Query(AndFilter(FieldFilter(FieldName,Phrase(VariableReference)),FieldFilter(FieldName,Comparison(Value(VariableReference)))))

# a variable inside a word is part of the word

host:web-$zone.example.com

==>

Query(FieldFilter(FieldName,Phrase))

# a name that goes on after the variable is a word

$a.b

==>

Query(Phrase)

# a dollar before a digit is no variable

$1

==>

Query(Phrase)

# a variable as a number

* | limit $n

==>

Query(MatchAll,LimitPipe(PipeName,VariableReference))

_time:$window offset $shift

==>

Query(TimeFilter(FieldName,VariableReference,TimeOffset(Keyword,VariableReference)))

# a variable as a field name

$field:error | fields $field

==>

Query(FieldFilter(FieldName(VariableReference),Phrase),FieldsPipe(PipeName,FieldName(VariableReference)))

# an uppercase name is no variable

$Env

* | stats by (_time:$__interval) count()
